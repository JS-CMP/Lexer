void benchmarkTokenization(const std::string& code, const std::string& name) {
    auto start = std::chrono::high_resolution_clock::now();

## 📚 Test Case Examples

### Basic Tokenization Test

```cpp
TEST(LexerTests, BasicTokenization) {
    std::string code = "var x = 42;";
    Lexer::Lexer lexer(code);
    auto tokens = lexer.tokenize();
    
    ASSERT_EQ(tokens.size(), 5);
    EXPECT_EQ(tokens[0].type, Lexer::TK_VAR);
    EXPECT_EQ(tokens[1].type, Lexer::TK_IDENTIFIER);
    EXPECT_EQ(tokens[1].value, "x");
    EXPECT_EQ(tokens[2].type, Lexer::TK_ASSIGN);
    EXPECT_EQ(tokens[3].type, Lexer::TK_NUMBER);
    EXPECT_EQ(tokens[3].value, "42");
    EXPECT_EQ(tokens[4].type, Lexer::TK_SEMICOLON);
}
```

### Error Handling Test

```cpp
TEST(LexerTests, ErrorHandling) {
    std::string invalidCode = "function @test() {}";
    Lexer::Lexer lexer(invalidCode);
    
    try {
        auto tokens = lexer.tokenize();
        FAIL() << "Expected exception for invalid code";
    } catch (const std::runtime_error& e) {
        EXPECT_STREQ(e.what(), "Invalid token '@'");
    }
}
```

### Streaming Mode Test

```cpp
TEST(LexerTests, StreamingMode) {
    std::string code = "const x = 1 + 2;";
    Lexer::Lexer lexer(code);
    
    int tokenCount = 0;
    while (true) {
        Lexer::Token token = lexer.nextToken();
        if (token.type == Lexer::TK_EOS) break;
        tokenCount++;
    }
    
    EXPECT_GT(tokenCount, 0);
}
```

### Performance Test Structure

```cpp
class PerformanceTests : public ::testing::Test {
protected:
    void SetUp() override {
        // Load test data
        smallCode = loadTestData("small.js");
        largeCode = loadTestData("large.js");
    }
    
    std::string smallCode;
    std::string largeCode;
};

TEST_F(PerformanceTests, BatchModeSpeed) {
    auto start = std::chrono::high_resolution_clock::now();
    Lexer::Lexer lexer(largeCode);
    auto tokens = lexer.tokenize();
    auto end = std::chrono::high_resolution_clock::now();
    
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
    EXPECT_LT(duration.count(), 100); // Should complete in <100ms
}

TEST_F(PerformanceTests, MemoryUsage) {
    // Test memory usage with large input
    std::string hugeCode(1024 * 1024, 'a'); // 1MB of 'a' characters
    
    Lexer::Lexer lexer(hugeCode);
    auto tokens = lexer.tokenize();
    
    // Streaming mode should use less memory
    size_t streamingMemory = measureMemoryUsage([&]() {
        Lexer::Lexer lexer(hugeCode);
        while (lexer.nextToken().type != Lexer::TK_EOS);
    });
    
    EXPECT_LT(streamingMemory, 1024 * 512); // <512KB
}
```

## 🔧 Test Utilities

### Custom Test Helpers

```cpp
// Helper to create lexer and tokenize
std::vector<Lexer::Token> tokenize(const std::string& code) {
    Lexer::Lexer lexer(code);
    return lexer.tokenize();
}

// Helper to find token by type
Lexer::Token findToken(const std::vector<Lexer::Token>& tokens, Lexer::TokenType type) {
    for (const auto& token : tokens) {
        if (token.type == type) {
            return token;
        }
    }
    return Lexer::Token(Lexer::TK_ILLEGAL);
}

// Helper to count tokens of specific type
int countTokens(const std::vector<Lexer::Token>& tokens, Lexer::TokenType type) {
    int count = 0;
    for (const auto& token : tokens) {
        if (token.type == type) {
            count++;
        }
    }
    return count;
}
```
